Graph ::= Node
Node  ::= PredicateNode | ConceptNode | UnknownNode

PredicateNode ::= "(" Variable "/" PropBankFrame Edge* ")"
ConceptNode   ::= "(" Variable "/" Concept Edge* ")"
UnknownNode   ::= "(" Variable "/" "amr-unknown" Edge* ")"

Edge          ::= ":" Role Node
PropBankFrame ::= Verb "-" Sense
Role          ::= ":" Relation Value
Value         ::= AMR | Variable | Literal
Concept       ::= Entity | Constant | Identifier
